/* "@" Notation Extension */

/* Extends argument to include "@" forms */
argument         ::= ...
                   | macro_string
                   | property_access
                   | variable_ref
                   | macro_auto

variable_ref     ::= "$" ( identifier | qualified_identifier )

property_access  ::= "@" function_call "." identifier { "." identifier }
                   | "@" qualified_identifier "." identifier { "." identifier }

macro_auto       ::= "@" ( identifier | qualified_identifier )

macro_string     ::= '@"' { ( string_char | "$" variable_ref ) } '"'

/* Extends function_call to allow property_access as base */
function_call    ::= ( identifier | property_access ) "(" [ arg_list ] ")"
