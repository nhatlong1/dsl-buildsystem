/* Parser Rules */

program          ::= { function_call }

function_call    ::= qualified_identifier "(" [ arg_list ] ")"
arg_list         ::= argument { "," argument }
argument         ::= identifier | literal | function_call

qualified_identifier ::= identifier { "." identifier }

identifier       ::= letter { letter | digit | "_" } 
                     - keyword_fn - reserved_constant

keyword_fn       ::= "DECLARE"
                   | "SET"
                   | "EXECUTE"
                   | "SELECT"
                   | "IF"
                   | "USING"
                   | "IMPORT"
                   | "BUILTIN"
                   | "EXTERNAL"
                   | "USER"

reserved_constant ::= "VARIABLE"
                    | "MACRO"
                    | "EXECUTABLE"

literal          ::= string
                   | template_string
                   | number
                   | boolean
                   | NULL
                   | reserved_constant

string           ::= '"' { string_char } '"'

template_string  ::= '"' { string_char | "$" number } '"'

string_char      ::= escape_sequence
                   | any_char - ( '"' | '\' )

escape_sequence  ::= '\' ( '"' | '\' | 'n' | 't' | 'r' | '0' 
                         | 'x' hex_digit hex_digit )

number           ::= "0" | ( digit - '0' ) { digit }

boolean          ::= "TRUE" | "FALSE"

NULL             ::= "NULL"

/* Lexical Rules */

letter           ::= "a".."z" | "A".."Z"
digit            ::= "0".."9"
hex_digit        ::= digit | "a".."f" | "A".."F"
any_char         ::= /* any printable character */
whitespace       ::= " " | "\t" | "\n" | "\r"

/* Lexer Notes */

comment          ::= Comments begin with "/*" and end with "*/".
                     Comments may not be nested.
